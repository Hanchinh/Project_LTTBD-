export const saveOrUpdateCustomerLocation = async (req: AuthenticatedRequest, res: Response) => {
  const { customerId } = req.params;
  const { latitude, longitude, gpsAccuracy, identificationNotes, imageUrl, imageBase64 } = req.body;
  const userId = req.user?.userId;

  if (latitude === undefined || longitude === undefined || !identificationNotes) {
    return res.status(400).json({
      success: false,
      message: 'Vui lÃ²ng cung cáº¥p Ä‘áº§y Ä‘á»§ latitude, longitude vÃ  ghi chÃº nháº­n diá»‡n',
    });
  }

  let createdFilePath: string | null = null;

  try {
    let finalImageUrl: string | null = imageUrl || null;

    // Process base64 photo if provided
    if (imageBase64 && typeof imageBase64 === 'string') {
      const matches = imageBase64.match(/^data:([A-Za-z-+\/]+);base64,(.+)$/);
      let buffer: Buffer;
      let ext = 'jpg';

      if (matches && matches.length === 3) {
        const mimeType = matches[1].toLowerCase();
        if (mimeType.includes('png')) ext = 'png';
        else if (mimeType.includes('webp')) ext = 'webp';
        else if (!mimeType.includes('jpeg') && !mimeType.includes('jpg')) {
          return res.status(400).json({
            success: false,
            message: 'Äá»‹nh dáº¡ng áº£nh khÃ´ng há»£p lá»‡. Chá»‰ cháº¥p nháº­n JPG, PNG, WEBP.',
          });
        }
        buffer = Buffer.from(matches[2], 'base64');
      } else {
        buffer = Buffer.from(imageBase64, 'base64');
      }

      // Check max size: 5MB
      if (buffer.length > 5 * 1024 * 1024) {
        return res.status(400).json({
          success: false,
          message: 'Dung lÆ°á»£ng áº£nh vÆ°á»£t quÃ¡ giá»›i háº¡n cho phÃ©p (tá»‘i Ä‘a 5MB)',
        });
      }

      // Generate randomized filename: never using customer name or phone
      const filename = `${crypto.randomUUID()}.${ext}`;
      const filePath = path.join(UPLOADS_DIR, filename);
      await fs.promises.writeFile(filePath, buffer);
      createdFilePath = filePath;

      finalImageUrl = `/api/v1/photos/${filename}`;
    }

    // Execute database operations in a strict single-client transaction
    const client = await pool.connect();
    try {
      await client.query('BEGIN');

      // Check if location exists
      const existing = await client.query('SELECT id, image_url FROM saved_locations WHERE customer_id = $1', [customerId]);

      let resultLocationId = '';

      if (existing.rows.length === 0) {
        // Create new location
        const insertRes = await client.query(
          `INSERT INTO saved_locations
           (customer_id, geom, latitude, longitude, gps_accuracy, identification_notes, image_url, created_by, updated_by)
           VALUES
           ($1, ST_SetSRID(ST_MakePoint($3, $2), 4326), $2, $3, $4, $5, $6, $7, $7)
           RETURNING id`,
          [customerId, latitude, longitude, gpsAccuracy || 0, identificationNotes, finalImageUrl, userId]
        );
        resultLocationId = insertRes.rows[0].id;
      } else {
        // If no new image was uploaded, preserve existing image_url
        if (!finalImageUrl) {
          finalImageUrl = existing.rows[0].image_url;
        }

        // Update existing location (Trigger automatically writes to location_history)
        const updateRes = await client.query(
          `UPDATE saved_locations
           SET geom = ST_SetSRID(ST_MakePoint($3, $2), 4326),
               latitude = $2,
               longitude = $3,
               gps_accuracy = $4,
               identification_notes = $5,
               image_url = $6,
               updated_by = $7,
               updated_at = CURRENT_TIMESTAMP
           WHERE customer_id = $1
           RETURNING id`,
          [customerId, latitude, longitude, gpsAccuracy || 0, identificationNotes, finalImageUrl, userId]
        );
        resultLocationId = updateRes.rows[0].id;
      }

      // Write audit log within the same transaction
      await client.query(
        `INSERT INTO audit_logs (user_id, action, entity, entity_id, details) VALUES ($1, $2, $3, $4, $5)`,
        [
          userId,
          existing.rows.length === 0 ? 'CREATE_LOCATION' : 'UPDATE_LOCATION',
          'SAVED_LOCATION',
          resultLocationId,
          JSON.stringify({
            customerId,
            latitude,
            longitude,
            gpsAccuracy,
            notes: identificationNotes,
            hasPhoto: !!finalImageUrl,
          }),
        ]
      );

      await client.query('COMMIT');

      return res.json({
        success: true,
        message: existing.rows.length === 0 ? 'LÆ°u vá»‹ trÃ­ khÃ¡ch hÃ ng thÃ nh cÃ´ng' : 'Cáº­p nháº­t vá»‹ trÃ­ khÃ¡ch hÃ ng thÃ nh cÃ´ng',

          locationId: resultLocationId,
          customerId,
          latitude,
          longitude,
          gpsAccuracy,
          identificationNotes,
          imageUrl: finalImageUrl,
        },
      });
    } catch (dbError: any) {
      await client.query('ROLLBACK');

      // If a new photo file was created on disk in this request, clean it up to prevent orphaned files
      if (createdFilePath && fs.existsSync(createdFilePath)) {

        }
      }

      throw dbError; // Rethrow to outer catch, preserving original error
    } finally {
      client.release();
    }
  } catch (error: any) {
    return res.status(500).json({ success: false, message: 'Lá»—i lÆ°u vá»‹ trÃ­ khÃ¡ch hÃ ng', error: error.message });

// Helper to normalize VN phone numbers
export function normalizePhoneNumber(raw: string): string {
  let clean = raw.replace(/\D/g, '');
  if (clean.startsWith('84') && clean.length === 11) {
    clean = '0' + clean.substring(2);
  } else if (clean.length === 9 && !clean.startsWith('0')) {
    clean = '0' + clean;
  }
  return clean;
}

export const createShift = async (req: AuthenticatedRequest, res: Response) => {
  const shipperId = req.user?.userId;
  const { workDate } = req.body;

  // Work date in Asia/Ho_Chi_Minh: YYYY-MM-DD
  const effectiveDate = workDate || new Date().toLocaleDateString('en-CA', { timeZone: 'Asia/Ho_Chi_Minh' });

  const client = await pool.connect();
  try {
    await client.query('BEGIN');

    // Count shifts on that date to generate sequence: SHF-YYYYMMDD-XXX
    const dateFormatted = effectiveDate.replace(/-/g, '');
    const countRes = await client.query(
      `SELECT COUNT(*)::int as count FROM shifts WHERE work_date = $1`,
      [effectiveDate]
    );
    const seq = (countRes.rows[0].count + 1).toString().padStart(3, '0');
    const shiftCode = `SHF-${dateFormatted}-${seq}`;

    const insertRes = await client.query(
      `INSERT INTO shifts (shift_code, work_date, shipper_id, status)
       VALUES ($1, $2, $3, 'PREPARING')
       RETURNING id, shift_code, work_date, status, created_at, updated_at`,
      [shiftCode, effectiveDate, shipperId]
    );

    const shift = insertRes.rows[0];

    // Log audit
    await client.query(
      `INSERT INTO audit_logs (user_id, action, entity, entity_id, details)
       VALUES ($1, 'CREATE_SHIFT', 'shifts', $2, $3)`,
      [shipperId, shift.id, JSON.stringify({ shiftCode, workDate: effectiveDate })]
    );

    await client.query('COMMIT');

    return res.status(201).json({
      success: true,
      message: 'Táº¡o ca lÃ m viá»‡c má»›i thÃ nh cÃ´ng',
      data: {
        id: shift.id,
        shiftCode: shift.shift_code,
        workDate: shift.work_date,
        status: shift.status,
        createdAt: shift.created_at,
        updatedAt: shift.updated_at,
        summary: {
          totalPoints: 0,
          verifiedCount: 0,
          temporaryCount: 0,
          noLocationCount: 0,
        },
      },
    });
  } catch (error: any) {
    await client.query('ROLLBACK');
    return res.status(500).json({
      success: false,
      message: 'Lá»—i táº¡o ca lÃ m viá»‡c má»›i',
      error: error.message,
    });
  } finally {
    client.release();

export const verifyPointLocation = async (req: AuthenticatedRequest, res: Response) => {
  const shipperId = req.user?.userId;
  const roleName = req.user?.roleName;
  const { id: shiftId, pointId } = req.params;
  const { latitude, longitude, gpsAccuracy, identificationNotes, imageBase64, imageUrl } = req.body;

  if (latitude === undefined || longitude === undefined || !identificationNotes) {
    return res.status(400).json({
      success: false,
      message: 'Vui lÃ²ng cung cáº¥p latitude, longitude vÃ  ghi chÃº nháº­n diá»‡n vá»‹ trÃ­',
    });
  }

  const lat = Number(latitude);
  const lng = Number(longitude);
  if (isNaN(lat) || isNaN(lng) || lat < -90 || lat > 90 || lng < -180 || lng > 180) {
    return res.status(400).json({
      success: false,
      message: 'Tá»a Ä‘á»™ WGS84 khÃ´ng há»£p lá»‡',
    });
  }

  let createdFilePath: string | null = null;

  try {
    let finalImageUrl: string | null = imageUrl || null;

    // Handle image upload if provided
    if (imageBase64 && typeof imageBase64 === 'string') {
      const matches = imageBase64.match(/^data:([A-Za-z-+\/]+);base64,(.+)$/);
      let buffer: Buffer;
      let ext = 'jpg';

      if (matches && matches.length === 3) {
        const mimeType = matches[1].toLowerCase();
        if (mimeType.includes('png')) ext = 'png';
        else if (mimeType.includes('webp')) ext = 'webp';
        buffer = Buffer.from(matches[2], 'base64');
      } else {
        buffer = Buffer.from(imageBase64, 'base64');
      }

      if (buffer.length > 5 * 1024 * 1024) {
        return res.status(400).json({
          success: false,
          message: 'Dung lÆ°á»£ng áº£nh vÆ°á»£t quÃ¡ 5MB',
        });
      }

      if (!fs.existsSync(UPLOADS_DIR)) {
        fs.mkdirSync(UPLOADS_DIR, { recursive: true });
      }

      const filename = `${crypto.randomUUID()}.${ext}`;
      const filePath = path.join(UPLOADS_DIR, filename);
      await fs.promises.writeFile(filePath, buffer);
      createdFilePath = filePath;
      finalImageUrl = `/api/v1/photos/${filename}`;
    }

    const client = await pool.connect();
    try {
      await client.query('BEGIN');

      // Check shift
      const shiftRes = await client.query(`SELECT id, shipper_id FROM shifts WHERE id = $1`, [shiftId]);
      if (shiftRes.rows.length === 0) {
        await client.query('ROLLBACK');
        return res.status(404).json({ success: false, message: 'KhÃ´ng tÃ¬m tháº¥y ca lÃ m viá»‡c' });
      }
      if (roleName === 'SHIPPER' && shiftRes.rows[0].shipper_id !== shipperId) {
        await client.query('ROLLBACK');
        return res.status(403).json({ success: false, message: 'Báº¡n khÃ´ng cÃ³ quyá»n sá»­a ca nÃ y' });
      }

      // Check point and get customerId
      const pointRes = await client.query(
        `SELECT id, customer_id, image_url FROM shift_delivery_points WHERE id = $1 AND shift_id = $2`,
        [pointId, shiftId]
      );
      if (pointRes.rows.length === 0) {
        await client.query('ROLLBACK');
        return res.status(404).json({ success: false, message: 'KhÃ´ng tÃ¬m tháº¥y Ä‘iá»ƒm giao trong ca' });
      }

      const customerId = pointRes.rows[0].customer_id;
      if (!finalImageUrl) {
        finalImageUrl = pointRes.rows[0].image_url;
      }

      // 1. Update point in shift to VERIFIED
      await client.query(

             gps_accuracy = $3,

             image_url = $5,

               identification_notes = $5,

        );
      }
